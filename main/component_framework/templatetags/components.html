 14@register.simple_tag(takes_context=True)
 15def live_component(context: dict, name: str, **params) -> str:
 16    """
 17    Render a live component.
 18
 19    Usage:
 20        {% load components %}
 21        {% live_component "counter" initial=5 %}
 22        {% live_component "order_form" pk=order.id %}
 23    """
 24    # Get component class
 25    component_cls = registry.get(name)
 26    if not component_cls:
 27        return f"<!-- Component '{name}' not found -->"
 28
 29    # Create component
 30    component = component_cls(**params)
 31
 32    # Dispatch (initial render)
 33    result = component.dispatch()
 34
 35    # Return HTML
 36    return mark_safe(result["html"])

 50@register.inclusion_tag("components/htmx_component.html", takes_context=True)
 51def htmx_component(context: dict, name: str, **params) -> dict:
 52    """
 53    Render component with HTMX wrapper.
 54
 55    Usage:
 56        {% htmx_component "counter" initial=5 %}
 57
 58    Requires template: components/htmx_component.html
 59    """
 60    component_cls = registry.get(name)
 61    if not component_cls:
 62        return {"error": f"Component '{name}' not found"}
 63
 64    component = component_cls(**params)
 65    result = component.dispatch()
 66
 67    return {
 68        "name": name,
 69        "html": mark_safe(result["html"]),
 70        "state": result["state"],
 71        "component_id": result["component_id"],
 72    }

 75@register.filter
 76def component_errors(errors_dict: dict | None, field_name: str) -> list:
 77    """
 78    Get field errors from component error dict.
 79
 80    Usage:
 81        {% if errors|component_errors:"email" %}
 82            {{ errors|component_errors:"email"|first }}
 83        {% endif %}
 84    """
 85    if isinstance(errors_dict, dict):
 86        return errors_dict.get(field_name, [])
 87    return []

 90@register.simple_tag
 91def component_js(component_id: str) -> str:
 92    """
 93    Generate JavaScript for component WebSocket connection.
 94
 95    Usage:
 96        {% component_js component.id %}
 97    """
 98    # Safely encode the component_id for embedding in JavaScript
 99    safe_id = json_module.dumps(str(component_id))
100    js = f"""
101    <script>
102    (function() {{
103        const componentId = {safe_id};
104        const proto = location.protocol === 'https:' ? 'wss://' : 'ws://';
105        const ws = new WebSocket(proto + location.host + '/ws/');
106
107        ws.onopen = function() {{
108            // Subscribe to component updates
109            ws.send(JSON.stringify({{
110                type: 'subscribe',
111                component_id: componentId
112            }}));
113        }};
114
115        ws.onmessage = function(event) {{
116            const data = JSON.parse(event.data);
117
118            if (data.type === 'component_update' && data.component_id === componentId) {{
119                // Update component HTML
120                const element = document.getElementById(componentId);
121                if (element) {{
122                    element.outerHTML = data.html;
123                }}
124            }}
125        }};
126
127        ws.onerror = function(error) {{
128            console.error('WebSocket error:', error);
129        }};
130
131        ws.onclose = function() {{
132            console.log('WebSocket closed');
133        }};
134    }})();
135    </script>
136    """
137    return mark_safe(js)

141@register.simple_tag(takes_context=True)
142def cotton_component(context: dict, name: str, **attrs) -> str:
143    """
144    Render a live component as a Cotton component.
145
146    Usage:
147        {% load components %}
148        {% cotton_component "counter" initial=5 class="my-counter" %}
149
150    This renders the component and wraps it in a Cotton-compatible structure.
151    """

153    if not component_cls:
154        return f"<!-- Component '{name}' not found -->"
155
156    # Create component
157    component = component_cls(**attrs)
158
159    # Dispatch
160    result = component.dispatch()
161
162    # For Cotton integration, we can pass props to Cotton components
163    # or just render the component directly
164    return mark_safe(result["html"])

167@register.simple_tag
168def component_attrs(component_id: str, state_dict: dict, **extra_attrs) -> str:
169    """
170    Generate HTMX attributes for component interaction.
171
172    Usage:
173        <button {% component_attrs component.id state %}>Click</button>
174    """
175    attrs = {
176        "hx-target": f"#{component_id}",
177        "hx-swap": "outerHTML",
178    }
179
180    # Add extra attributes
181    attrs.update(extra_attrs)
182
183    # Build attribute string with HTML escaping to prevent XSS
184    attr_str = " ".join(f'{escape(str(k))}="{escape(str(v))}"' for k, v in attrs.items())
185    return mark_safe(attr_str)
