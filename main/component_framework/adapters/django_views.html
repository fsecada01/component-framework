 20@require_POST
 21def component_view(request: HttpRequest, name: str) -> JsonResponse:
 22    """
 23    Django view for component endpoints.
 24
 25    POST /components/{name}/
 26    Body (JSON or form-data):
 27        - event: Event name
 28        - payload: Event data (JSON string)
 29        - state: Serialized state (JSON string)
 30        - params: Component parameters (JSON string)
 31
 32    Returns:
 33        JSON response with html, state, and component_id
 34    """
 35    try:
 36        # Get component class
 37        component_cls = registry.get(name)
 38        if not component_cls:
 39            return JsonResponse({"error": f"Component '{name}' not found"}, status=404)
 40
 41        # Check component-level permissions
 42        for perm_class in getattr(component_cls, "permission_classes", []) or [AllowAny]:
 43            if not perm_class().has_permission(request, component_cls):
 44                return JsonResponse({"error": "Permission denied"}, status=403)
 45
 46        # Parse request data (supports both JSON and form-encoded)
 47        if request.content_type == "application/json":
 48            try:
 49                data = json.loads(request.body)
 50            except json.JSONDecodeError as e:
 51                return JsonResponse({"error": f"Invalid JSON: {e}"}, status=400)
 52        else:
 53            data = {
 54                "event": request.POST.get("event"),
 55                "payload": request.POST.get("payload", "{}"),
 56                "state": request.POST.get("state"),
 57                "params": request.POST.get("params", "{}"),
 58            }
 59
 60        # Extract parameters
 61        event = data.get("event")
 62        payload_str = data.get("payload", "{}")
 63        state_str = data.get("state")
 64        params_str = data.get("params", "{}")
 65
 66        # Parse JSON strings
 67        try:
 68            payload = json.loads(payload_str) if isinstance(payload_str, str) else payload_str
 69            params = json.loads(params_str) if isinstance(params_str, str) else params_str
 70        except json.JSONDecodeError as e:
 71            return JsonResponse({"error": f"Invalid JSON in payload/params: {e}"}, status=400)
 72
 73        # Deserialize state
 74        state = None
 75        if state_str:
 76            try:
 77                state = StateSerializer.deserialize(state_str)
 78            except Exception as e:
 79                return JsonResponse({"error": f"Invalid state: {e}"}, status=400)
 80
 81        # Compute optimistic patch BEFORE dispatch (uses pre-dispatch state).
 82        # Hydrate state so get_optimistic_patch can access current state values.
 83        optimistic_patch: dict | None = None
 84        if event:
 85            probe = component_cls(**params)
 86            if state:
 87                probe.hydrate(state)
 88            else:
 89                probe.mount()
 90            optimistic_patch = probe.get_optimistic_patch(event, payload or {})
 91
 92        # Create and dispatch component
 93        component = component_cls(**params)
 94        result = component.dispatch(event=event, payload=payload, state=state)
 95
 96        # Serialize state for response
 97        result["state"] = StateSerializer.serialize(result["state"])
 98
 99        # Include optimistic patch if the component provided one
100        if optimistic_patch is not None:
101            result["optimistic"] = optimistic_patch
102
103        return JsonResponse(result)
104
105    except Exception:
106        logger.exception(f"Error processing component '{name}'")
107        return JsonResponse({"error": "Internal server error"}, status=500)

133class ComponentView(View):
134    """
135    Class-based view for component endpoints.
136
137    Usage:
138        # urls.py
139        from component_framework.adapters.django_views import ComponentView
140
141        urlpatterns = [
142            path("components/<str:name>/", ComponentView.as_view()),
143        ]
144
145    Or with custom configuration:
146        class MyComponentView(ComponentView):
147            http_method_names = ['post', 'get']  # Allow GET too
148
149            def get_component_params(self, request, **kwargs):
150                # Custom param extraction
151                params = super().get_component_params(request, **kwargs)
152                params['user_id'] = request.user.id
153                return params
154    """
155
156    http_method_names = ["post"]
157
158    def check_component_permissions(
159        self, request: HttpRequest, component_cls: type[Component]
160    ) -> JsonResponse | None:
161        """
162        Check component-level permission_classes.
163
164        Returns a JsonResponse (403) if any permission class denies access,
165        or None if all permissions are granted.
166        """
167        for perm_class in getattr(component_cls, "permission_classes", []) or [AllowAny]:
168            if not perm_class().has_permission(request, component_cls):
169                return JsonResponse({"error": "Permission denied"}, status=403)
170        return None
171
172    def post(self, request: HttpRequest, name: str, **kwargs) -> JsonResponse:
173        """Handle POST request for component."""
174        try:
175            # Get component class
176            component_cls = self.get_component_class(name)
177            if not component_cls:
178                return self.component_not_found(name)
179
180            # Check component-level permissions
181            perm_response = self.check_component_permissions(request, component_cls)
182            if perm_response is not None:
183                return perm_response
184
185            # Parse request data
186            data = self.parse_request_data(request)
187
188            # Extract component parameters
189            event = data.get("event")
190            payload = self.parse_payload(data.get("payload", "{}"))
191            state = self.parse_state(data.get("state"))
192            params = self.parse_params(data.get("params", "{}"))
193
194            # Add custom params
195            params.update(self.get_component_params(request, **kwargs))
196
197            # Compute optimistic patch BEFORE dispatch (uses pre-dispatch state).
198            # A separate probe instance is used so the real dispatch starts clean.
199            optimistic_patch: dict | None = None
200            if event:
201                probe = self.create_component(component_cls, params)
202                if state:
203                    probe.hydrate(state)
204                else:
205                    probe.mount()
206                optimistic_patch = probe.get_optimistic_patch(event, payload)
207
208            # Create and dispatch component
209            component = self.create_component(component_cls, params)
210            result = self.dispatch_component(component, event, payload, state)
211
212            # Post-process result
213            result = self.post_process_result(result, component)
214
215            # Attach optimistic patch if the component provided one
216            if optimistic_patch is not None:
217                result["optimistic"] = optimistic_patch
218
219            return self.render_response(result)
220
221        except Exception as e:
222            return self.handle_error(e, name)
223
224    def get_component_class(self, name: str) -> type[Component] | None:
225        """Get component class from registry."""
226        return registry.get(name)
227
228    def parse_request_data(self, request: HttpRequest) -> dict:
229        """Parse request data (JSON or form-encoded)."""
230        if request.content_type == "application/json":
231            try:
232                return json.loads(request.body)
233            except json.JSONDecodeError as e:
234                raise ValueError(f"Invalid JSON: {e}")
235        else:
236            return {
237                "event": request.POST.get("event"),
238                "payload": request.POST.get("payload", "{}"),
239                "state": request.POST.get("state"),
240                "params": request.POST.get("params", "{}"),
241            }
242
243    def parse_payload(self, payload_str: str | dict) -> dict:
244        """Parse payload JSON string."""
245        if isinstance(payload_str, dict):
246            return payload_str
247        try:
248            return json.loads(payload_str)
249        except json.JSONDecodeError as e:
250            raise ValueError(f"Invalid payload JSON: {e}")
251
252    def parse_state(self, state_str: str | None) -> dict | None:
253        """Parse state JSON string."""
254        if not state_str:
255            return None
256        try:
257            return StateSerializer.deserialize(state_str)
258        except Exception as e:
259            raise ValueError(f"Invalid state: {e}")
260
261    def parse_params(self, params_str: str | dict) -> dict:
262        """Parse params JSON string."""
263        if isinstance(params_str, dict):
264            return params_str
265        try:
266            return json.loads(params_str)
267        except json.JSONDecodeError as e:
268            raise ValueError(f"Invalid params JSON: {e}")
269
270    def get_component_params(self, request: HttpRequest, **kwargs) -> dict:
271        """
272        Get additional component parameters.
273
274        Override to inject custom parameters (user, session data, etc.)
275        """
276        return {}
277
278    def create_component(self, component_cls: type[Component], params: dict) -> Component:
279        """Create component instance."""
280        return component_cls(**params)
281
282    def dispatch_component(
283        self, component: Component, event: str | None, payload: dict, state: dict | None
284    ) -> dict:
285        """Dispatch component with event."""
286        return component.dispatch(event=event, payload=payload, state=state)
287
288    def post_process_result(self, result: dict, component: Component) -> dict:
289        """Post-process component result. Override to add custom data."""
290        result["state"] = StateSerializer.serialize(result["state"])
291        return result
292
293    def render_response(self, result: dict) -> JsonResponse:
294        """Render JSON response."""
295        return JsonResponse(result)
296
297    def component_not_found(self, name: str) -> JsonResponse:
298        """Handle component not found."""
299        return JsonResponse({"error": f"Component '{name}' not found"}, status=404)
300
301    def handle_error(self, error: Exception, name: str) -> JsonResponse:
302        """Handle errors."""
303        logger.exception(f"Error processing component '{name}'")
304        return JsonResponse({"error": "Internal server error"}, status=500)

394class ComponentPageView(TemplateView):
395    """
396    Template view that renders a page with components.
397
398    Usage:
399        class MyPageView(ComponentPageView):
400            template_name = "my_page.html"
401            components = {
402                "counter": {"initial": 5},
403                "form": {"user_id": 123},
404            }
405
406            def get_component_params(self, component_name):
407                # Override to add dynamic params
408                params = super().get_component_params(component_name)
409                if component_name == "form":
410                    params["user_id"] = self.request.user.id
411                return params
412    """
413
414    components: dict[str, dict] = {}
415
416    def get_context_data(self, **kwargs):
417        """Add component rendering to context."""
418        context = super().get_context_data(**kwargs)
419        context["components"] = {}
420
421        for component_name, params in self.get_components().items():
422            component_cls = registry.get(component_name)
423            if component_cls:
424                # Add custom params
425                params = self.get_component_params(component_name, params)
426
427                # Render component
428                component = component_cls(**params)
429                result = component.dispatch()
430                context["components"][component_name] = result
431
432        return context
433
434    def get_components(self) -> dict[str, dict]:
435        """Get components to render. Override for dynamic components."""
436        return self.components
437
438    def get_component_params(self, component_name: str, params: dict) -> dict:
439        """Get parameters for a specific component. Override to add custom params."""
440        return params.copy()

446class CacheMixin:
447    """
448    Mixin to add caching to component responses via Django's cache framework.
449
450    Caches the JSON result of component dispatches (mount/hydrate only). On
451    subsequent requests with the same component name, params, and state, the
452    cached result is returned directly without re-dispatching the component.
453
454    Cache bypass rules:
455    - **Event requests** (requests that carry an ``event`` field) are never
456      served from or written to the cache. Events mutate component state, so
457      caching them would cause stale state to be returned on the next request.
458    - **Non-200 responses** are not stored in the cache. Errors, 404s, and
459      permission denials must always go through the full request cycle.
460
461    Attributes:
462        cache_timeout: Cache TTL in seconds. Defaults to 60.
463
464    Customising the cache key:
465        Override ``get_cache_key`` to produce application-specific keys::
466
467            class MyComponentView(CacheMixin, ComponentView):
468                cache_timeout = 300  # 5 minutes
469
470                def get_cache_key(self, name, params, state):
471                    return f"component:{name}:{params.get('id')}"
472
473    MRO note:
474        Always place ``CacheMixin`` *before* ``ComponentView`` in the class
475        bases so that this ``post`` override runs first::
476
477            class CachedView(CacheMixin, ComponentView):
478                pass
479    """
480
481    cache_timeout: int = 60
482
483    def get_cache_key(self, name: str, params: dict, state: dict | None) -> str:
484        """Generate cache key. Override for custom keys."""
485        import hashlib
486
487        params_json = json.dumps(params, sort_keys=True)
488        state_json = json.dumps(state or {}, sort_keys=True)
489        key_data = f"{name}:{params_json}:{state_json}"
490        return f"component:{hashlib.md5(key_data.encode()).hexdigest()}"
491
492    def post(self, request: HttpRequest, name: str, **kwargs) -> JsonResponse:
493        """Check cache before processing. Event requests bypass the cache entirely."""
494        from django.core.cache import cache
495
496        # Parse request data — methods provided by ComponentView via MRO
497        data = self.parse_request_data(request)  # type: ignore[unresolved-attribute]  # cooperative mixin
498
499        # Event requests mutate state and must never be served from or stored
500        # in the cache. Skip all cache logic and go straight to the parent view.
501        if data.get("event"):
502            return super().post(request, name, **kwargs)  # type: ignore[unresolved-attribute]  # cooperative mixin
503
504        params = self.parse_params(data.get("params", "{}"))  # type: ignore[unresolved-attribute]  # cooperative mixin
505        state = self.parse_state(data.get("state"))  # type: ignore[unresolved-attribute]  # cooperative mixin
506
507        cache_key = self.get_cache_key(name, params, state)
508        cached_result = cache.get(cache_key)
509
510        if cached_result is not None:
511            return JsonResponse(cached_result)
512
513        # Process normally
514        response = super().post(request, name, **kwargs)  # type: ignore[unresolved-attribute]  # cooperative mixin
515
516        # Only cache successful responses
517        if response.status_code == 200:
518            cache.set(cache_key, json.loads(response.content), self.cache_timeout)
519
520        return response
